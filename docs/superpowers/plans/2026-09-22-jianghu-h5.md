```ts
    this.lastOptions = [];

      return;
    }
    const found = this.program.filter((b) => b.id === target);
    this.runBehaviors(found);
  }
```

- [ ] **Step 3: 实现 binding.ts**

```ts
import type { Engine, RenderCommand } from "@core";
import type { OptionItem, Scene } from "@core";

export type LineKind = "narration" | "dialog" | "monologue" | "voiceoff";
export interface Line { kind: LineKind; npc: string | null; text: string; }
export interface UiState { scene: Scene | null; fx: string | null; lines: Line[]; options: OptionItem[]; }

export class GameBinding {
  readonly engine: Engine;
  readonly state: UiState = { scene: null, fx: null, lines: [], options: [] };
  private optionsEmitted = false;

  constructor(engine: Engine) {
    this.engine = engine;
    engine.onCmd = (cmd) => this.handle(cmd);
  }

  start(): void { this.engine.start(); }

  triggerPoint(id: string): void {
    this.state.options = [];
    this.engine.triggerPoint(id);
  }

  pick(label: string): void {
    this.optionsEmitted = false;
    const ok = this.engine.choose(label);
    if (ok && !this.optionsEmitted) this.state.options = [];
  }

  private handle(cmd: RenderCommand): void {
    switch (cmd.t) {
      case "scene":
        this.state.scene = cmd.scene;
        this.state.fx = cmd.scene.fx ?? null;
        this.state.lines = [];
        this.state.options = [];
        this.optionsEmitted = false;
        break;
      case "fx": this.state.fx = cmd.kind; break;
      case "narration": this.state.lines.push({ kind: "narration", npc: null, text: cmd.text }); break;
      case "dialog": this.state.lines.push({ kind: "dialog", npc: cmd.npc, text: cmd.text }); break;
      case "monologue": this.state.lines.push({ kind: "monologue", npc: null, text: cmd.text }); break;
      case "voiceoff": this.state.lines.push({ kind: "voiceoff", npc: null, text: cmd.text }); break;
      case "options":
        this.state.options = cmd.options;
        this.optionsEmitted = true;
        break;
    }
  }
}
```

> 说明：`pick` 在 `choose` 前重置 `optionsEmitted`；若 goto 目标本身带新选项，`options` 指令会在 `choose` 期间同步触发 → `optionsEmitted=true`，则保留 `state.options` 不被清空。

```vue
<script setup lang="ts">
import { ref, watch, onUnmounted } from "vue";
import type { Line } from "../engine/binding";

const props = defineProps<{ line: Line; active: boolean }>();
const shown = ref(props.active ? 0 : props.line.text.length);
const revealed = ref(!props.active);
let timer: ReturnType<typeof setInterval> | undefined;

function animate() {
  if (revealed.value) return;
  timer = setInterval(() => {
    shown.value += 1;
    if (shown.value >= props.line.text.length) {
      clearInterval(timer); timer = undefined;
      revealed.value = true;
    }
  }, 40);
}

watch(() => props.active, (a) => { if (a) animate(); }, { immediate: true });

function reveal() {
  if (timer) { clearInterval(timer); timer = undefined; }
  shown.value = props.line.text.length;
  revealed.value = true;
}

onUnmounted(() => { if (timer) clearInterval(timer); });
</script>

<template>
  <div class="dialog-slot" @click="reveal">
    <span v-if="line.npc" class="npc-tag">{{ line.npc }}</span>
    <p class="line-text">{{ line.text.slice(0, shown) }}</p>
  </div>
</template>
```
