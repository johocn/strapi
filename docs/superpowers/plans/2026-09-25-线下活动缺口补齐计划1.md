// plugins/zhao-point/server/src/services/seed-rules.ts
/**
 * 种子规则收集（纯逻辑）：从插件默认配置的 increaseRules/decreaseRules 与
 * DB 已有 action 列表求差集。补种语义是「只增不改」——同名 action 已存在即跳过，
 * 生产人工调整过的规则不会被覆盖。
 */
export type RuleDef = {
  points?: number;
  limitPerDay?: number;
  limitPerUser?: number;
  limitPerDayPerUser?: number;
  isOneTime?: boolean;
  description?: string;
  taskGroup?: string;
  extraConfig?: Record<string, unknown>;
};

export type MissingRule = {
  action: string;
  category: "increase" | "decrease";
  rule: RuleDef;
};

export const collectMissingRules = (
  increaseRules: Record<string, RuleDef> | undefined,
  decreaseRules: Record<string, RuleDef> | undefined,
  existingActions: string[]
): MissingRule[] => {
  const existing = new Set(existingActions);
  const missing: MissingRule[] = [];
  for (const [action, rule] of Object.entries(increaseRules || {})) {
    if (!existing.has(action)) missing.push({ action, category: "increase", rule });
  }
  for (const [action, rule] of Object.entries(decreaseRules || {})) {
    if (!existing.has(action)) missing.push({ action, category: "decrease", rule });
  }
  return missing;
};
```

/**
 * 幂等收紧票据 token 为 NOT NULL（一次性凭据防重放底线）。
 * 有 NULL 行时告警跳过，不阻断启动。
 */
const ensureTicketTokenNotNull = async (strapi: Core.Strapi) => {
  const knex = strapi.db.connection;
  const row = await knex(TICKET_TABLE).whereNull("token").count({ n: "*" }).first();
  const n = Number((row as any)?.n ?? 0) || 0;
  if (n > 0) {
    strapi.log.warn(`[zhao-point] ${TICKET_TABLE}.token 有 ${n} 行空值，跳过 NOT NULL，请先人工核账`);
    return;
  }
  await knex.raw(`ALTER TABLE ${TICKET_TABLE} ALTER COLUMN token SET NOT NULL`);
  strapi.log.info(`[zhao-point] 票据 token 已收紧为 NOT NULL (${TICKET_TABLE})`);
};
```

      } else {

- [ ] **Step 4.1: 写手册（内容骨架如下，每步含 目标/操作/期望 三段）**

1. **前置**：管理端 admin 登录 h.joho.cn 活动管理；C 端 2-3 个 SSO 测试用户已登录 H5（JWT 从已登录会话请求头取）；演练活动参数：标题`【演练】勿报名-20260925`、渠道=平台根渠道、容量 2、免费、报名窗=现在起 10 分钟、活动开始时间=+30 分钟、formConfig 最简（姓名 1 字段）。
2. **S1 满员报名**：用户 A、B 报名 → 均 `kind:"active"`；SQL `SELECT status, count(*) FROM activity_signups GROUP BY status` 出现 active=2。
3. **S2 候补**：用户 C 报名 → `waitlisted:true, position:1`。
4. **S3 双击并发**：用户 A 同 JWT 两个并发 `signup` curl → 恰一条成功，另一条 `dup` 或幂等；`activity_signups` 无重复行（部分唯一索引兜底）。
5. **S4 取消+递补**：A `cancel` → C `promoted`，转正站内信出现。
6. **S5 并发取消 drill**：重新满员+候补，A/B 两并发 `cancel` → 仅一次 promote，C 只收一条转正通知、signups 只一行 active。
7. **S6 取消后重报**：A 再报名 → 新行 active（旧行 cancelled），部分唯一索引不误伤。
8. **S7 扫码核销**：管理端把活动时间改为过去 → C 端出码（票据 `{token,expiresAt}`）→ 管理端 scan.vue 或 curl `scan-checkin` → `ok:true`；`activity_attendances.method='worker_scan'` 且 `signups.attended_at` 同步非空（两视图同源核对）。
9. **S8 票据幂等**：同 token 二次提交 → `ticket_used`。
10. **S9 票据过期**：重新出码等 5 分钟提交 → `ticket_expired`。
11. **S10 自助+手动**：C 端 `checkin {method:"self"}` 成功；管理端 manual 核销（不填理由 → `reason_required`；填 ≥2 字符理由 → 成功且审计有 operator/记录）。
12. **S11 名单导出**：管理端导出 CSV → 三种取值（已到场/未到场/已取消）与 DB 一致，formConfig 动态列出现。
13. **S12 积分账实**：`SELECT user_id, SUM(amount) FROM point_records GROUP BY user_id` 与余额快照一致（演练账号），报名/到场/取消往返积分逐笔对得上。
14. **S13 通知**：站内信核对 waitlisted/confirm/promoted/before 四类。
15. **S14 收尾**：演练活动下架（unpublish）；验收记录表填入每步实测结果（截图/SQL 输出）。
