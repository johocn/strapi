# www.joho.cn 静态站活动页 实施计划（Plan B）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 www.joho.cn 静态站新建线下活动详情页 `/activity/{documentId}` 与列表页 `/activities`，通过标签联动展示相关活动/文章/案例/产品/FAQ/课程/教程（构建期静态写入），报名状态/名额客户端实时。

**Architecture:** Next.js App Router 静态导出（与 geo-article 同模式）：构建期 fetch zhao-point 公开 API + zhao-tag 聚合接口 → 静态 HTML；客户端组件轮询实时报名状态。无新增依赖。

**Tech Stack:** Next.js（App Router + output export）、TypeScript、CSS（现有 globals.css 变量）

依赖：Plan A 完成后端（聚合接口、活动 tags relation、slug/documentId 可用）、`docs/superpowers/specs/2026-09-08-活动页标签联动-design.md`。

**关键决策：** 活动 URL 段用 `documentId`（迁移后 slug=documentId，稳定唯一）；活动不 i18n（schema localized:false），仅默认语言路由。

```ts
import { API_ROOT } from "@/lib/env";

export type ActivityCard = {
  documentId: string;
  slug?: string;
  title: string;
  description?: string;
  startTime?: string;
  endTime?: string;
  venueName?: string;
  capacity?: number;
  usedCapacity?: number;
  status?: string;
  coverImage?: { url?: string } | null;
  tags?: { documentId: string; name?: string; color?: string }[];
  lecturer?: { name?: string; position?: string } | null;
  promoModules?: any[];
  customPromoHtml?: string;
  customPromoActive?: boolean;
};

export type RelatedGroup = Record<string, { type: string; documentId: string; title: string; summary: string; url: string }[]>;

/** 活动详情（构建期静态） */
export async function getActivity(documentId: string): Promise<{ status: number; activity: ActivityCard | null }> {
  try {
    const res = await fetch(`${API_ROOT}/activities/${encodeURIComponent(documentId)}`);
    if (!res.ok) return { status: res.status, activity: null };
    return { status: res.status, activity: await res.json() };
  } catch {
    return { status: 500, activity: null };
  }
}

/** 构建期枚举已发布活动 documentId（generateStaticParams） */
export async function listActivityDocumentIds(): Promise<string[]> {
  try {
    const res = await fetch(`${API_ROOT}/activities?pageSize=200`);
    if (!res.ok) return [];
    const data = await res.json();
    const results = Array.isArray(data) ? data : data?.results ?? [];
    return results.map((r: any) => r?.documentId).filter(Boolean);
  } catch {
    return [];
  }
}

/** 活动列表（构建期，已发布全部） */
export async function listActivities(): Promise<ActivityCard[]> {
  try {
    const res = await fetch(`${API_ROOT}/activities?pageSize=100`);
    if (!res.ok) return [];
    const data = await res.json();
    const results = Array.isArray(data) ? data : data?.results ?? [];
    return results.filter((r: any) => r?.status !== "draft" && r?.status !== "archived");
  } catch {
    return [];
  }
}

/** 标签聚合（构建期，静态写入 HTML） */
export async function getRelatedByTags(tagIds: string[]): Promise<RelatedGroup> {
  if (!tagIds.length) return {};
  try {
    const res = await fetch(`${API_ROOT}/zhao-tag/v1/related-by-tags?tags=${tagIds.join(",")}&limit=3`);
    if (!res.ok) return {};
    const data = await res.json();
    return data?.data ?? {};
  } catch {
    return {};
  }
}
```

> 注意：`API_ROOT` 经 `/api` 前缀反代（`https://www.joho.cn/api`），与 geo-article 数据层一致。`/api/zhao-tag/v1/related-by-tags` 需构建前确认 nginx 反代覆盖（与 `/api/zhao-website/` 同规则）。

- [ ] **Step 1: 详情页路由（[slug]=documentId）**

创建 `app/(default)/activity/[slug]/page.tsx`：

```tsx
import type { Metadata } from "next";
import { SITE_URL } from "@/lib/env";
import { getActivity, getRelatedByTags } from "@/lib/activity";
import ActivityView from "@/components/views/ActivityView";

export const dynamic = "error"; // 静态导出：禁止运行时动态

export async function generateStaticParams() {
  const { listActivityDocumentIds } = await import("@/lib/activity");
  return (await listActivityDocumentIds()).map((documentId) => ({ slug: documentId }));
}

export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
  const { slug } = await params;
  const { activity } = await getActivity(slug);
  if (!activity) return { title: "活动不存在" };
  return {
    title: activity.title,
    description: activity.description?.slice(0, 150) || undefined,
    alternates: { canonical: `${SITE_URL}/activity/${activity.documentId}` },
  };
}

export default async function ActivityPage({ params }: { params: Promise<{ slug: string }> }) {
  const { slug } = await params;
  const { activity } = await getActivity(slug);
  if (!activity) return <main className="geo-page"><p>活动不存在或已下线</p></main>;

  const tagIds = (activity.tags ?? []).map((t) => t.documentId);
  const related = await getRelatedByTags(tagIds);

  return <ActivityView activity={activity} related={related} />;
}
```

创建 `components/modules/activity/ActivityCta.tsx`（`"use client"`）：

```tsx
"use client";
import { useEffect, useState } from "react";
import type { ActivityCard } from "@/lib/activity";

export default function ActivityCta({ activity }: { activity: ActivityCard }) {
  const [live, setLive] = useState(activity);
  const [error, setError] = useState(false);

  useEffect(() => {
    let timer: ReturnType<typeof setInterval>;
    const poll = async () => {
      try {
        const res = await fetch(`/api/zhao-point/v1/activities/${activity.documentId}`, { cache: "no-store" });
        if (!res.ok) throw new Error(String(res.status));
        setLive(await res.json());
        setError(false);
      } catch {
        setError(true); // 保留快照，非阻断
      }
    };
    poll();
    timer = setInterval(poll, 30000);
    const onVis = () => { if (document.visibilityState === "visible") poll(); };
    document.addEventListener("visibilitychange", onVis);
    return () => { clearInterval(timer); document.removeEventListener("visibilitychange", onVis); };
  }, [activity.documentId]);

  const used = live.usedCapacity ?? 0;
  const cap = live.capacity ?? 0;
  const left = cap ? cap - used : 0;
  const signupUrl = `https://v.joho.cn/pages/activity/detail?id=${activity.documentId}`;

  return (
    <section className="activity-cta">
      {error ? <p className="activity-live-tip">实时状态获取失败，以下为最近数据</p> : null}
      {cap ? <p className="activity-slots">{left > 0 ? `剩余名额 ${left}` : "已报满"}</p> : null}
      <div className="activity-cta-actions">
        <a className={`geo-cta ${left === 0 ? "disabled" : ""}`} href={left > 0 ? signupUrl : "#"} aria-disabled={left === 0}>
          {left > 0 ? "去报名" : "已报满"}
        </a>
        <a className="geo-cta ghost" href={signupUrl}>活动详情与咨询</a>
      </div>
    </section>
  );
}
```
